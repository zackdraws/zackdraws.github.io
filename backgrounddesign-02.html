async function includeExternalContent() { try { const response = await fetch('/css/bg-list.html'); // Replace with your file path if (!response.ok) throw new Error('Network response was not ok'); const data = await response.text(); document.getElementById('content-container').innerHTML = data; } catch (error) { console.error('Error fetching content:', error); } } includeExternalContent();
